npm install antifraude-pro
import { AntiFraudePro } from 'antifraude-pro'; // 1. Coleta instantânea de sinais em memória (0ms de latência) const sinais = await AntiFraudePro.coletarSinaisNavegador(); const { deviceId } = AntiFraudePro.gerarDeviceId(sinais); // 2. Envia para sua API de pagamento com o payload fetch('/api/checkout', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ valor: 149.90, cpf: '52998224725', email: 'cliente@gmail.com', deviceId, sinais }) });
